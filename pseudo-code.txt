buchberger_simple(basis):
    flag = true
    
    while flag:
        additives = []
        flag = false
        for f in basis:
            for g in (basis - {f}):
                s = S(f, g) mod basis
                if s != 0:
                    additives.append(s)
                    flag = true
        basis = basis + additives



crit(basis, B, i, j):
    f = basis[i]
    g = basis[j]
    for k in range(len(B)):
        if k == i or k == j:
            continue
        h = basis[k]
        if find(B, (i, k)) == -1 and find(B, (j, k)) == -1 and LCM(LT(f), LT(g)) mod LT(h) == 0:
            return true
    return false


buchberger(basis):
    t = len(basis)
    B = []
    G = copy(basis)

    for i in range(t-1):
        for j in range(i + 1, t):
            B.append((i, j))

    while len(B) != 0:
        k = rand(len(B))
        i, j = B[k]
        f = basis[i]
        g = basis[j]

        if LCM(LT(f), LT(g)) != LT(f)LT(g) and not crit(basis, B, i, j):
            s = S(f, g) mod basis
            if s != 0:
                
                basis += s
                for i in range(t):
                    B += (i, t)

                t = t + 1
        B -= (i, j)

form(P, F):
    for i in range(P):
        if crit(i, len(P), F):
            P += (i, len(P))

GMI(F, P, h):
    P' = [(f(i), h) for i in range(len(P))]

    for (f, g) in P:
        L = lcm(f, g)
        if lt(h) | L and lcm(h, f) != L and lcm(h, g) != L:
            P = P\{f, g}

    for (f, h) in P':
        for (g, h) in P'\{(f, h)}:
            if lcm(f, h) | lcm(g, h):
                P' = P'\{(g, h)}

    for (f, h) in P':
        if gcd(f, h) == 1:
            P' = P'\{(f, h)}

    F = F + h;
    P = P + P'
